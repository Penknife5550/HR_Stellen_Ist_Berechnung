-- Lookup: gültiger Stundensatz für einen Schlüssel zu einem Stichtag.
-- Liefert NULL, wenn kein Satz greift (vom Aufrufer prüfen!).
CREATE OR REPLACE FUNCTION get_stundensatz(
    p_schlüssel SMALLINT,
    p_datum     DATE DEFAULT CURRENT_DATE
) RETURNS NUMERIC(6,2)
LANGUAGE sql STABLE AS $$
    SELECT stundensatz_euro
    FROM   vergütungsstufen
    WHERE  pedav_schlüssel = p_schlüssel
      AND  gültig_ab <= p_datum
      AND  (gültig_bis IS NULL OR gültig_bis >= p_datum)
    ORDER BY gültig_ab DESC
    LIMIT 1;
$$;


-- ============================================================
-- 2. ADAPTER  Untis-SalaryPeHour  ->  PEDAV-Schlüssel   (NEU)
--    >>> EINZIGE Stelle, an der die 200/400-Deutung steht. <<<
--    Tabelle statt CASE im Code: fachlich pflegbar ohne Deploy.
-- ============================================================

CREATE TABLE IF NOT EXISTS salary_per_hour_mapping (
    salary_per_hour     INTEGER     PRIMARY KEY,   -- Roh-Wert aus Teacher.SalaryPeHour
    pedav_schlüssel     SMALLINT    NULL           -- NULL = kein Schlüssel ableitbar
        CHECK (pedav_schlüssel IS NULL OR pedav_schlüssel BETWEEN 0 AND 4),
    bemerkung           TEXT        NULL
);

INSERT INTO salary_per_hour_mapping (salary_per_hour, pedav_schlüssel, bemerkung)
VALUES
    (400, 4, 'ANNAHME: 400 = Schlüssel 4 (A13 hD, GY/BK). 71 Lehrkräfte.'),
    (200, 2, 'ANNAHME: 200 = Schlüssel 2 (A12 gD, GS). 69 Lehrkräfte.'),
    (0, NULL, 'Nicht gepflegt -> KEINE Abrechnung möglich. 25 Lehrkräfte. Sekretariat pflegen lassen.')
ON CONFLICT (salary_per_hour) DO NOTHING;

-- Übersetzt einen SalaryPeHour-Rohwert in den PEDAV-Schlüssel.
-- Unbekannter Wert -> Exception (lieber lauter Fehler als stille Falschabrechnung).
CREATE OR REPLACE FUNCTION map_salary_per_hour_to_schluessel(p_salary INTEGER)
RETURNS SMALLINT
LANGUAGE plpgsql STABLE AS $$
DECLARE
    v_schlüssel SMALLINT;
    v_treffer   INTEGER;
BEGIN
    SELECT pedav_schlüssel, 1 INTO v_schlüssel, v_treffer
    FROM   salary_per_hour_mapping
    WHERE  salary_per_hour = p_salary;

    IF v_treffer IS NULL THEN
        RAISE EXCEPTION
          'Unbekannter SalaryPeHour-Wert: %. In salary_per_hour_mapping ergänzen.', p_salary;
    END IF;

    RETURN v_schlüssel;  -- darf NULL sein (= 0-Fall, nicht gepflegt)
END;
$$;

    substitution_id     BIGINT       NOT NULL PRIMARY KEY,

-- 5. BERECHNUNG (Vollzeit-Pfad, monatlich) -- vollständig lauffähig
--    Teilzeit-Pfad (wochenweise, anteilige Bagatellgrenze ab 07.06.2025)
--    ist als TODO markiert: braucht (a) Teilzeitquote je LK und
--    (b) verifizierte Art-Flags. Bewusst nicht geraten.

-- 5.1 Monatssumme der vergütungsfähigen Vertretungsstunden je Lehrkraft.
--     Vollzeit-Logik der Rechtsdoku: IST aus Vertretungen minus Freisetzungen.
CREATE OR REPLACE VIEW v_mehrarbeit_substitution_monat AS
WITH bewertet AS (
    SELECT
        s.teacher_id_subst                         AS teacher_id,
        s.schoolyear_id,
        date_trunc('month', s.datum)::date         AS monat,
        -- Vertretungen zählen positiv, Freisetzungen negativ (Vollzeitregel)
        CASE
            WHEN s.art = 'VERTRETUNG'  THEN COALESCE(s.subst_value, 0)
            WHEN s.art = 'FREISETZUNG' THEN -COALESCE(s.subst_value, 0)
            ELSE 0                                 -- ENTFALL u.a.: nicht vergütungsrelevant
        END                                        AS wert_stunden
    FROM stg_untis_substitution s
    WHERE s.deleted = FALSE
      AND s.teacher_id_subst IS NOT NULL
)
SELECT
    teacher_id,
    schoolyear_id,
    monat,
    SUM(wert_stunden) AS mehrarbeit_netto_stunden   -- vor Bagatellgrenze/Kappung
FROM bewertet
GROUP BY teacher_id, schoolyear_id, monat;


-- 5.2 Anwendung von Bagatellgrenze, Monats-/Jahresobergrenze und Euro-Betrag.
--     Vollzeitregeln (Rechtsdoku §1.4): Bagatelle 3 h, Kappung 24 h/Monat.
--     Jahresgrenze 288 h: hier als Warnspalte; harte Kappung im ETL ergänzen.
CREATE OR REPLACE VIEW v_mehrarbeit_abrechnung_vollzeit AS
SELECT
    m.teacher_id,
    t.p_number,
    m.monat,
    m.mehrarbeit_netto_stunden,
    -- Schlüssel aus SalaryPeHour (bestätigte Entscheidung)
    map_salary_per_hour_to_schluessel(t.salary_per_hour)                       AS schlüssel,
    -- Bagatellgrenze 3 h + Kappung 24 h -> vergütungsfähige Stunden
    CASE
        WHEN m.mehrarbeit_netto_stunden < 3 THEN 0
        ELSE LEAST(m.mehrarbeit_netto_stunden, 24)
    END                                                                        AS stunden_vergütet,
    -- Stundensatz zum Monatsende (Satz gilt für den Abrechnungsmonat)
    get_stundensatz(
        map_salary_per_hour_to_schluessel(t.salary_per_hour),
        (m.monat + INTERVAL '1 month - 1 day')::date
    )                                                                          AS stundensatz_euro,
    -- Betrag = vergütungsfähige Stunden x Satz
    CASE
        WHEN m.mehrarbeit_netto_stunden < 3 THEN 0::numeric
        ELSE LEAST(m.mehrarbeit_netto_stunden, 24)
             * get_stundensatz(
                 map_salary_per_hour_to_schluessel(t.salary_per_hour),
                 (m.monat + INTERVAL '1 month - 1 day')::date)
    END                                                                        AS betrag_euro,
    -- Kennzeichen 'V' nur bei tatsächlicher Vergütung
    CASE WHEN m.mehrarbeit_netto_stunden >= 3 THEN 'V' ELSE NULL END           AS kennzeichen,
    -- Defensive Warnungen (statt stiller Fehler)
    CASE WHEN t.salary_per_hour = 0 OR t.salary_per_hour IS NULL
         THEN 'WARNUNG: kein Vergütungsschlüssel (SalaryPeHour=0) -> nicht abrechenbar'
         WHEN m.mehrarbeit_netto_stunden > 24
         THEN 'HINWEIS: Monatsobergrenze 24 h erreicht, Rest gekappt'
         ELSE NULL END                                                         AS hinweis
FROM v_mehrarbeit_substitution_monat m
JOIN stg_untis_teacher t
  ON t.teacher_id = m.teacher_id
 AND t.schoolyear_id = m.schoolyear_id
WHERE t.deleted = FALSE;
